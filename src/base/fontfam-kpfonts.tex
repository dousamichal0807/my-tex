% Auxiliary macros
\edef \KpFonts@@seriffam {Roman}
\edef \KpFonts@@regularweight {Regular}
\edef \KpFonts@@italicstyle {Italic}
\def \KpFonts@@publicmacros {%
  \serif \sans \mono
  \upright \italic
  \lightserif \regularserif \semiboldserif \boldserif \regularsans \boldsans
  \light \regular \semibold \bold
  \smallcaps \nosmallcaps
  \mathaltscript \mathstdscript
  \currfontname \currmathfontname \fontcleanup}
\def\KpFonts@@makepublic#1{%
  \ea\let\ea#1\csname KpFonts@\csstring#1\endcsname}

%===========%
% Modifiers %
%===========%

% Sub-families
\def \KpFonts@serif {%
  \def \KpFonts@@weight {\KpFonts@@serifweight}%
  \edef \KpFonts@@subfam {Roman}}
\def \KpFonts@sans {%
  \def \KpFonts@@weight {\KpFonts@@sansweight}%
  \edef \KpFonts@@subfam {Sans}}
\def \KpFonts@mono {%
  \def \KpFonts@@weight {\KpFonts@@sansweight}%
  \edef \KpFonts@@subfam {Mono}}

% Style variants
\def \KpFonts@upright       {\edef \KpFonts@@style {}}
\def \KpFonts@italic        {\edef \KpFonts@@style {Italic}}

% Serif font weight variants
\def \KpFonts@lightserif    {\edef \KpFonts@@serifweight {Light}}
\def \KpFonts@regularserif  {\edef \KpFonts@@serifweight {Regular}}
\def \KpFonts@semiboldserif {\edef \KpFonts@@serifweight {Semibold}}
\def \KpFonts@boldserif     {\edef \KpFonts@@serifweight {Bold}}

% Sans-serif & monospace font weight variants
\def \KpFonts@regularsans   {\edef \KpFonts@@sansweight {Regular}}
\def \KpFonts@boldsans      {\edef \KpFonts@@sansweight {Bold}}

% Font weight variants
\def \KpFonts@light         {\KpFonts@regularsans \KpFonts@lightserif}
\def \KpFonts@regular       {\KpFonts@regularsans \KpFonts@regularserif}
\def \KpFonts@semibold      {\KpFonts@boldsans    \KpFonts@semiboldserif}
\def \KpFonts@bold          {\KpFonts@boldsans    \KpFonts@boldserif}

% Alternate script math
\def \KpFonts@mathaltscript {\edef \KpFonts@@mathscript {+ss01;}}
\def \KpFonts@mathstdscript {\edef \KpFonts@@mathscript {}}

% \currfontname
\def \KpFonts@currfontname {%
  [Kp\KpFonts@@subfam-%
  \ea\ifx \KpFonts@@weight\KpFonts@@regularweight
    \ifx \KpFonts@@style\KpFonts@@italicstyle Italic%
    \else Regular\fi
  \else
    \KpFonts@@weight\KpFonts@@style
  \fi]:}
0
% \currmathfontname
\def \KpFonts@currmathfontname {%
  [KpMath-%
  \ifx \KpFonts@@subfam\KpFonts@@seriffam
    \KpFonts@@weight
  \else
    Sans%
    \ifx \KpFonts@@sansweight\KpFonts@@regularweight
    \else \KpFonts@@weight \fi
  \fi]:\KpFonts@@mathscript}

% Cleanup
\def \KpFonts@fontcleanup {%
  \ea\foreach\ea{\KpFonts@@publicmacros}\undef}

% Selector
\def \KpFonts {%
  \fontcleanup
  \ea\foreach\ea{\KpFonts@@publicmacros}\KpFonts@@makepublic}

% Defaults
\KpFonts@serif \KpFonts@upright \KpFonts@regular \KpFonts@nosmallcaps
\KpFonts@mathstdscript
