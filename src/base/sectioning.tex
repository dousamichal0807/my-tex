% Counters
\newcount \chapternum       \chapternum=0
\newcount \sectionnum       \sectionnum=0
\newcount \subsectionnum    \subsectionnum=0

% Current chapter, section and subsection number
\let \currchapternum \empty
\let \currsectionnum \empty
\let \currsubsectionnum \empty

% Chapter to appendix switching
\let \ifappendix=\iffalse
\def \appendices {\chapternum=0 \let\ifappendix\iftrue}

% \every{chapter, section, subsection} registers
\newtoks \everychapter
\newtoks \everysection
\newtoks \everysubsection

% Allow suppression of numbering of the next chapter, section or subsection
\newif \iftocnum
\newif \iftocshow
\def \resettocflags {\tocnumtrue \tocshowtrue}
\resettocflags

% Formatting of chapter, section and subsection numbers
\def \thechapternum {%
  \ifappendix \Athe\chapternum
  \else \the\chapternum \fi}
\def \thesectionnum {%
  \ifx \currchapternum\empty \else \currchapternum.\fi
  \the\sectionnum}
\def \thesubsectionnum {%
  \ifx \currsectionnum\empty \else \currsectionnum.\fi
  \the\subsectionnum}

% Resetting counters
\def \resetsectionnum {\global\sectionnum=0\resetsubsectionnum}
\def \resetsubsectionnum {\global\subsectionnum=0\relax}

% Increasing counters
\def \incrchapternum {\global\advance \chapternum by 1\relax \resetsectionnum}
\def \incrsectionnum {\global\advance \sectionnum by 1\relax \resetsubsectionnum}
\def \incrsubsectionnum {\global\advance \subsectionnum by 1\relax}

% The "public" macros; the optional parameter is the label that can be
% referenced
\def \chapter {\readoptional\@chapterA}
\def \section {\readoptional\@sectionA}
\def \subsection {\readoptional\@subsectionA}

% Chapter macro where most things are done
\def \@chapterreftag {\ifappendix appendix\else chapter\fi}
\def \@chapterA #1\par {\@dosectioning{chapter}{\@chapterreftag}{#1}}
\def \@sectionA #1\par {\@dosectioning{section}{section}{#1}}
\def \@subsectionA #1\par {\@dosectioning{subsection}{subsection}{#1}}

% #1 ... type of sectioning (chapter, section, subsection)
% #2 ... *.ref file tag (appendix, chapter, section, subsection)
% #3 ... the title
\def \@dosectioning #1#2#3{%
  % Auxiliary \lets and \defs
  \expandafter\let \expandafter\@sect@incrnum \csname incr#1num\endcsname
  \expandafter\let \expandafter\@sect@thenum  \csname the#1num\endcsname
  \expandafter\let \expandafter\@sect@print   \csname print#1\endcsname
  \expandafter\let \expandafter\@sect@every   \csname every#1\endcsname
  % Get current sectioning element number
  \let \@sect@currnum \empty
  \iftocnum \@sect@incrnum \edef\@sect@currnum{\@sect@thenum}\fi
  \expandafter\glet \csname curr#1num\endcsname \@sect@currnum
  % Get *.ref file label
  \let \@sect@currlabel \theoptional
  \ifx \@sect@currlabel\undefined  \let\@sect@currlabel\empty  \fi
  % Do \every#1
  \the \@sect@every
  % Do the printing
  \@sect@print{\@sect@currnum}{#3}%
  % Write to *.ref file
  \writeref{\@sect@currlabel}{#2\iftocshow\else*\fi}{\@sect@currnum}{#3}%
  % Reset flags
  \resettocflags}
