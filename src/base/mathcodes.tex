\newcount \charcount
\newcount \charno
\newcount \mathcharatomtype
\newcount \mathcharfam
\newcount \mathcharno

\def \incrchardef #1{\chardef#1\charno \advance\charno1\relax}

% Sets \Umathcode based on values of \charno, \mathcharatomtype, \mathcharfam
% and \mathcharno 
\def \setregmathcode {\Umathcode\charno \mathcharatomtype\mathcharfam\mathcharno \relax}

\def \bulksetmathcode {%
  \ifnum \charcount>\z@
    \setregmathcode
    \advance \charcount -1%
    \advance \charno 1%
    \advance \mathcharno 1%
    \relax \bulksetmathcode
  \fi}

% Sets the "math alternative" of given character. Syntax:
%   \setmathchar <character> <math character codepoint>.
% Math font family number and math atom type are taken from \mathcharfam and
% \mathcharatomtype. 
\def \setmathchar {\afterassignment\setmathcharA \charno=}
\def \setmathcharA {\afterassignment\setregmathcode \mathcharno=}

% Defines given control sequence to be a mathemathical character. Syntax:
%     \definemathchar <control sequence> <math atom type> <math character codepoint>
% Math font family number is taken from \mathcharfam. 
\def \definemathchar #1{\def\@mathcharname{#1}\afterassignment\definemathcharA \mathcharatomtype=}
\def \definemathcharA {\afterassignment\definemathcharB \mathcharno=}
\def \definemathcharB {\ea\Umathchardef\@mathcharname\mathcharatomtype\mathcharfam\mathcharno\relax}

% Sets the "math alternative" of given character. Syntax:
%   \setmathcode <character codepoint> <atom type> <math character codepoint>.
% Math font family number is taken from \mathcharfam. 
\def \setmathcode {\afterassignment\setmathcodeA \charno=}
\def \setmathcodeA {\afterassignment\setmathcodeB \mathcharatomtype=}
\def \setmathcodeB {\afterassignment\setregmathcode \mathcharno=}

% Lower and upper greek letters as they appear in Unicode range for Greek and
% Coptic (i.e. starting from U+00391 and U+003B1).
\def \lowergreekletters {%
  \alpha \beta \gamma \delta \epsilon \zeta \eta \theta \iota \kappa \lambda \mu
  \nu \xi \omicron \pi \rho \finsigma \sigma \tau \upsilon \phi \chi \psi \omega}
\def \uppergreekletters {%
  \Alpha \Beta \Gamma \Delta \Epsilon \Zeta \Eta \Theta \Iota \Kappa \Lambda \Mu
  \Nu \Xi \Omicron \Pi \Rho \unused \Sigma \Tau \Upsilon \Phi \Chi \Psi \Omega}

\charno="00391 \ea\foreach \ea{\uppergreekletters} \incrchardef
\charno="003B1 \ea\foreach \ea{\lowergreekletters} \incrchardef
\let \unused=\@@undefined
