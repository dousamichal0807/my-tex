\newcount \charcount
\newcount \charno
\newcount \mathcharatomtype
\newcount \mathcharno

\def \setregmathcode {%
  \Umathcode\charno \mathcharatomtype\z@\mathcharno \relax}

\def \bulksetmathcode {%
  \ifnum \charcount>\z@
    \setregmathcode
    \advance\charcount\m@ne
    \advance\charno\@ne
    \advance\mathcharno\@ne
    \bulksetmathcode
  \fi}

\def \setmathchar {\afterassignment\setmathchar@A \charno=}
\def \setmathchar@A {\afterassignment\setregmathcode \mathcharno=}

\def \definemathchar #1{%
  \def\@mathcharname{#1}%
  \afterassignment\definemathcharA \mathcharatomtype=}
\def \definemathcharA {%
  \afterassignment\definemathcharB \mathcharno=}
\def \definemathcharB {%
  \expandafter\Umathchardef \@mathcharname
  \mathcharatomtype\z@\mathcharno \relax}

\def \setmathcode {\afterassignment\setmathcode@A \charno=}
\def \setmathcode@A {\afterassignment\setmathcode@B \mathcharatomtype=}
\def \setmathcode@B {\afterassignment\setregmathcode \mathcharno=}

\def \setdelcode {\afterassignment\setdelcode@A \charno=}
\def \setdelcode@A {\afterassignment\regsetdelcode \mathcharno=}

\def \regsetdelcode {\Udelcode\charno \z@\mathcharno}

% Lower and upper greek letters as they appear in Unicode range for Greek and
% Coptic (i.e. starting from U+00391 and U+003B1).

\def \lowergreekletters {%
  \alpha \beta \gamma \delta \epsilon \zeta \eta \theta \iota \kappa \lambda \mu
  \nu \xi \omicron \pi \rho \finsigma \sigma \tau \upsilon \phi \chi \psi \omega}
\def \uppergreekletters {%
  \Alpha \Beta \Gamma \Delta \Epsilon \Zeta \Eta \Theta \Iota \Kappa \Lambda \Mu
  \Nu \Xi \Omicron \Pi \Rho \@@unused \Sigma \Tau \Upsilon \Phi \Chi \Psi \Omega}

\def \incrchardef #1{\chardef#1\charno \advance\charno1\relax}
\charno="00391 \ea\foreach \ea{\uppergreekletters} \incrchardef
\charno="003B1 \ea\foreach \ea{\lowergreekletters} \incrchardef
\undef\@@unused
