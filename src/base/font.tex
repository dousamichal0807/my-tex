% Load LuaTeX OpenType library
\input luaotfload.sty

%===========%
% Font size %
%===========%

\newdimen \currfontsize
\currfontsize=10pt
\baselineskip=12pt

%===============%
% Font features %
%===============%

% Default ligatures (liga)
\def \enablefontligs {\def\@ffliga{}}
\def \disablefontligs {\def\@ffliga{-liga;}}

% Small capitals (smcp)
\def \enablesmallcaps {\def\@ffsmcp{+smcp;}}
\def \disablesmallcaps {\def\@ffsmcp{}}

% Artificial bold
\def \emboldenfont #1{\def\@ffembolden{embolden=#1;}}
\def \noemboldenfont {\def\@ffembolden{}}

% Ligatures implemented by luaotfload.sty (tlig), same as in Computer Modern
% fonts
\def \enabletexligs {\def\@fftlig{+tlig;}}
\def \disabletexligs {\def\@fftlig{}}

% Default settings
\enablefontligs \enabletexligs \disablesmallcaps \noemboldenfont

% Suppressing unwanted ligatures
\def \suppresslig {\nobreak\null\nobreak}

%===========================%
% \fontcleanup, \selectfont %
%===========================%

\def \fontpubmacros {\currmathfontname \currfontname \fontcleanup}

\def \selectfont {%
  \font \@currfont=\currfontname \@ffliga \@fftlig \@ffsmcp \@ffembolden \space at\currfontsize
  \@currfont}

\chardef \mainmathfam=0

\def \selectmathfont {%
  \font \@tenmath=\currmathfontname \@ffembolden mode=base;script=math;ssty=0;mathsize=1; at\currfontsize
  \font \@sevenmath=\currmathfontname \@ffembolden mode=base;script=math;ssty=1;mathsize=2; at\currfontsize
  \font \@fivemath=\currmathfontname \@ffembolden mode=base;script=math;ssty=2;mathsize=3; at\currfontsize
  \textfont\mainmathfam=\@tenmath
  \scriptfont\mainmathfam=\@sevenmath
  \scriptscriptfont\mainmathfam=\@fivemath
  \fam=\mainmathfam\relax}
