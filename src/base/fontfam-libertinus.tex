%===========%
% Modifiers %
%===========%

% Sub-family
\def \Libertinus@serif      {\edef\Libertinus@@subfam{Serif}}
\def \Libertinus@sans       {\edef\Libertinus@@subfam{Sans}}
\def \Libertinus@mono       {\edef\Libertinus@@subfam{Mono}}

% Font weight
\def \Libertinus@regular    {\edef\Libertinus@@weight{Regular}}
\def \Libertinus@semibold   {\edef\Libertinus@@weight{Semibold}}
\def \Libertinus@bold       {\edef\Libertinus@@weight{Bold}}

% Italic font
\def \Libertinus@upright    {\edef\Libertinus@@style{}}
\def \Libertinus@italic     {\edef\Libertinus@@style{Italic}}

% Defaults
\Libertinus@serif \Libertinus@regular \Libertinus@upright

%===========%
% Font name %
%===========%

% Auxiliary definitions
\def \Libertinus@@regularweight {Regular}
\def \Libertinus@@italicstyle {Italic}
\def \Libertinus@@fontvariant {\Libertinus@@weight \Libertinus@@style}

% Normal font
\def \Libertinus@currfontname {%
  [Libertinus\Libertinus@@subfam-%
  % Regular+Italic -> only Italic
  \ifx \Libertinus@@weight\Libertinus@@regularweight
    \ifx \Libertinus@@style\empty \Libertinus@@regularweight
    \else \Libertinus@@italicstyle \fi
  \else \Libertinus@@weight\Libertinus@@style \fi]:}

% Math font
\def \Libertinus@currmathfontname {%
  [LibertinusMath-Regular]:%
  \ifx \Libertinus@@weight\Libertinus@@regularweight \else
    embolden=1.7;\fi}

%============%
% The switch %
%============%

% Auxiliary definitions
\def \Libertinus@makepublic #1{%
  \ea\let \ea#1\csname Libertinus@\csstring#1\endcsname}
\def \Libertinus@publicmacros {%
  \serif \sans \mono \regular \semibold \bold \upright \italic \currfontname
  \currmathfontname \fontcleanup}

% Font setup & cleanup
\def \Libertinus@fontcleanup {%
  \ea\foreach \ea{\Libertinus@publicmacros}\undef}
\def \Libertinus@fontsetup {%
  \ea\foreach \ea{\Libertinus@publicmacros}\Libertinus@makepublic}

% The switch
\def \Libertinus {\fontcleanup \Libertinus@fontsetup}
