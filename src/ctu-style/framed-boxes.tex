\newbox \CTU@framedsiderulebox
\newbox \CTU@framedbox

%----------%
% Settings %
%----------%

\newdimen \CTU@framedsiderulewidth    \CTU@framedsiderulewidth=2pt
\newdimen \CTU@framedhpadding         \CTU@framedhpadding=8pt
\newdimen \CTU@framedvpadding         \CTU@framedvpadding=5pt
\newdimen \CTU@framedinnerpadding     \CTU@framedinnerpadding=8pt
\def \CTU@framedheaderfont {}
\def \CTU@framedcontentfont {}

%----------------------------------%
% \CTU@beginframed, \CTU@endframed %
%----------------------------------%

\def \CTU@beginframed {%
  \skiplines 1
  \beginenv{CTU@framed}%
  % Save the material inside \CTU@framedbox
  \setbox \CTU@framedbox=\vtop \bgroup
    % Make content narrower by the \CTU@framedsiderulewidth, \CTU@framedhpadding
    % from both sides, 2\baselineskip for the pictogram and \CTU@framedinnerpadding
    % for space between the framed and the pictogram:
    \advance \hsize by -\CTU@framedsiderulewidth
    \advance \hsize by -2\CTU@framedhpadding
    \advance \hsize by -2\baselineskip
    \advance \hsize by -\CTU@framedinnerpadding
    % See \strut in defs/layout.tex
    \par\noindent \strut {\CTU@framedheaderfont \CTU@framedlabel}%
    \endgraf \CTU@framedcontentfont \firstnoindent}

\def \CTU@endframed {%
    % Finish \CTU@framedbox from \CTU@beginframed
    \endgraf \vskip -\baselineskip
    \par\leavevmode \strut\hfill \endgraf
  \egroup
  % Depth of \CTU@framedbox must be at least 1.25 * b:
  \ifdim \dp\CTU@framedbox<1.25\baselineskip
    \dp\CTU@framedbox=1.25\baselineskip
  \fi
  \ht\CTU@framedbox=0.75\baselineskip
  % Create the vertical rule, \CTU@framedvpadding higher and deeper than \CTU@framedbox.
  % Calculate vertical rule height (\tmpdimenA) and depth (\tmpdimenB)
  \tmpdimenA=\ht\CTU@framedbox \advance \tmpdimenA by \CTU@framedvpadding
  \tmpdimenB=\dp\CTU@framedbox \advance \tmpdimenB by \CTU@framedvpadding
  % Save the vertical rule inside \CTU@framedsiderulebox together with rule for
  % background color:
  \setbox \CTU@framedsiderulebox=\hbox to \hsize{%
    % The side rule
    \CTU@framedsiderulecolor
    \vrule height \tmpdimenA depth \tmpdimenB width \CTU@framedsiderulewidth \relax
    % The background
    \CTU@framedbgcolor
    \leaders \hrule height \tmpdimenA depth \tmpdimenB \hfill
    % Reset color
    \Black}%
  % Reset height and depth of \CTU@framedsiderulebox:
  \dp\CTU@framedsiderulebox=\z@
  \ht\CTU@framedsiderulebox=\z@
  % Calculate total height (\tmpdimenA := \ht + \dp) of \CTU@framedbox
  \tmpdimenA=\ht\CTU@framedbox
  \advance\tmpdimenA by \dp\CTU@framedbox \relax
  % Assemble final box
  \setbox \tmpboxA=\hbox{%
    \box \CTU@framedsiderulebox
    \hskip -\hsize \hskip \CTU@framedsiderulewidth \hskip \CTU@framedhpadding
    \hbox to 2\baselineskip {\CTU@framedsiderulecolor \CTU@framedicon \hss}%
    \hskip \CTU@framedinnerpadding
    \box \CTU@framedbox}%
  % Output the box
  \ht\tmpboxA=\z@ \dp\tmpboxA=\z@
  \par\noindent \leavevmode \lower 0.5\baselineskip \hbox{\box\tmpboxA} \endgraf
  \nobreak\vskip \tmpdimenA
  \nobreak\vskip -\baselineskip
  \nobreak\null
  \endenv{CTU@framed}%
  \firstnoindent\skiplines 1 }

%------------------%
% \begin... macros %
%------------------%

\def \CTU@beginnote {%
  \beginenv{CTU@note}%
  \let \CTU@framedicon=\CTU@noteicon
  \let \CTU@framedbgcolor=\CTU@BlueLLLL
  \let \CTU@framedsiderulecolor=\CTU@Blue
  \let \CTU@framedlabel=\CTU@notelabel
  \CTU@beginframed}

\def \CTU@beginwarn {%
  \beginenv{CTU@warn}%
  \let \CTU@framedicon=\CTU@warnicon
  \let \CTU@framedsiderulecolor=\CTU@Orange
  \let \CTU@framedbgcolor=\CTU@OrangeLLLL
  \let \CTU@framedlabel=\CTU@warnlabel
  \CTU@beginframed}

%----------------%
% \end... macros %
%----------------%

\def \CTU@endnote {\CTU@endframed \endenv{CTU@note}}
\def \CTU@endwarn {\CTU@endframed \endenv{CTU@warn}}

%==================%
% Framed equations %
%==================%

\newcount \CTU@framedeqlines
\newbox   \CTU@framedeqbox
\newdimen \CTU@framedeqboxht

\def \CTU@beginframedeq {%
  \endgraf \beginenv{CTU@framedeq}%
  \afterassignment\CTU@beginframedeqA
  \CTU@framedeqlines=}
\def \CTU@beginframedeqA {%
  \skiplines 1 \nobreak\null \nobreak\vskip-\baselineskip
  \CTU@framedeqboxht \the\CTU@framedeqlines\baselineskip
  \advance \CTU@framedeqboxht -\baselineskip
  \advance \CTU@framedeqboxht \topskip
  \setbox\CTU@framedeqbox \vtop to\CTU@framedeqboxht \bgroup
    \advance \hsize by-2\CTU@framedhpadding
    \advance \hsize by-\CTU@framedsiderulewidth
    \null\vss \leavevmode\strut \hfill \null $$}

\def \CTU@endframedeq {%
  $$\strut\endgraf \vss \skiplines 1 \leavevmode\strut\endgraf \egroup
  \setbox\CTU@framedeqbox \hbox{%
    {\gray[0.95]\vrule width\hsize
      height\dimexpr \topskip+\CTU@framedvpadding
      depth\dimexpr \the\CTU@framedeqlines\baselineskip
        -\baselineskip+\CTU@framedvpadding \relax}%
    \hskip -\hsize
    {\CTU@BlackLL \vrule width\CTU@framedsiderulewidth
      height\dimexpr \topskip+\CTU@framedvpadding
      depth\dimexpr \the\CTU@framedeqlines\baselineskip
        -\baselineskip+\CTU@framedvpadding \relax}%
    \hskip\CTU@framedhpadding \box\CTU@framedeqbox \hskip\CTU@framedhpadding}%
  \ht\CTU@framedeqbox\z@ \dp\CTU@framedeqbox\z@
  \noindent \lower.1\baselineskip \box\CTU@framedeqbox \endgraf
  \nobreak \skiplines \CTU@framedeqlines
  \nobreak \vskip -2\baselineskip
  \nobreak \null
  \endenv{CTU@framedeq}%
  \skiplines 1\relax}
