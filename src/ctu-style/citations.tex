\def \CTU@begincitations {%
  \vfill \null \nobreak\vfill
  \beginenv{CTU@citations}%
  \vbox\bgroup
    \leftskip=0pt plus 1fil
    \rightskip=0pt \parfillskip=0pt\relax}
\def \CTU@endcitations {\egroup\endenv{CTU@citations}\eject}

\def \CTU@citeauthorfont {}
\def \CTU@citebibreffont {}
\def \CTU@citecontentfont {}
\def \CTU@citenamefont {}
\def \CTU@citeleftquotes {``}
\def \CTU@citerightquotes {''}

\def \begincite #1 \from[#2][#3] #4: #5 \endcite {%
  \skiplines 1
  {\CTU@citecontentfont
    \CTU@citeleftquotes#1\hbox to\z@{\CTU@citerightquotes \hss}}%
  \break
  {\CTU@citeauthorfont --- \ignorespaces#4:}
  {\CTU@citenamefont \ignorespaces#5}\break
  {\CTU@citebibreffont \bib[#2][#3]}\Black\strut
  \endgraf}
